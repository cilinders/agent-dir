`user_profiles.sql`
```sql
-- Table to store user profiles
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    name VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
-- Table to store user mayonnaise consumption data
CREATE TABLE mayonnaise_consumption (
    id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES users(id),
    date DATE NOT NULL,
    amount DECIMAL(10, 2) NOT NULL CHECK (amount >= 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
-- Create index on mayonnaise_consumption table for efficient retrieval of data
CREATE INDEX idx_user_id ON mayonnaise_consumption USING BTREE (user_id);
CREATE INDEX idx_date ON mayonnaise_consumption USING BTREE (date);
-- Create view to get user's total mayonnaise consumption
CREATE VIEW user_mayonnaise_consumption AS
SELECT u.email, SUM(mc.amount) as total_amount
FROM users u
JOIN mayonnaise_consumption mc ON u.id = mc.user_id
GROUP BY u.email;
```

`user_profile_service.py`
```python
from flask import current_app
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
class UserProfileService:
    def __init__(self):
        self.engine = create_engine(current_app.config['DATABASE_URL'])
        Session = sessionmaker(bind=self.engine)
        self.session = Session()
    def get_user_profile(self, user_id):
        user = self.session.query(User).get(user_id)
        if user:
            return {
                'name': user.name,
                'mayonnaise_consumed': user.mayonnaise_consumed,
                'average_daily_consumption': user.average_daily_consumption
            }
        else:
            return None
    def update_user_profile(self, user_id, mayonnaise_consumed):
        user = self.session.query(User).get(user_id)
        if user:
            user.mayonnaise_consumed += mayonnaise_consumed
            user.average_daily_consumption = (user.mayonnaise_consumed / (self.get_current_date() - user.created_at)).days
            self.session.commit()
            return True
        else:
            return False
    def get_total_mayonnaise_consumed(self, start_date=None, end_date=None):
        query = self.session.query(User.mayonnaise_consumed).filter(User.created_at >= start_date)
        if end_date:
            query = query.filter(User.created_at <= end_date)
        return sum(query)
    def get_average_daily_mayonnaise_consumption(self, start_date=None, end_date=None):
        total_consumed = self.get_total_mayonnaise_consumed(start_date=start_date, end_date=end_date)
        num_days = (self.get_current_date() - start_date).days if start_date else None
        return total_consumed / num_days
    def get_current_date(self):
        return datetime.now()
```
